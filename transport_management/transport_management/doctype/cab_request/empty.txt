import frappe
from frappe.model.document import Document
from frappe.utils import now_datetime, get_datetime
from frappe import _


class CabRequest(Document):

    # 🔒 DATE & TIME VALIDATION
    def validate(self):
        if self.booking_datetime:
            if get_datetime(self.booking_datetime) < now_datetime():
                frappe.throw(_("Booking cannot be in the past. Please select a future time."))


# 🚫 Prevent Employee from changing status
def prevent_employee_status_change(doc, method):

    if "Employee" in frappe.get_roles():
        old_doc = doc.get_doc_before_save()

        if old_doc and doc.status != old_doc.status:
            frappe.throw(_("Employees are not allowed to change status."))


# 🚖 Only Driver can mark Completed
def validate_driver_complete(doc, method):

    old_doc = doc.get_doc_before_save()

    if not old_doc:
        return

    if old_doc.status != doc.status and doc.status == "Completed":
        if "Driver" not in frappe.get_roles():
            frappe.throw(_("Only Driver can mark the trip as Completed."))


# 🔁 Copy Allocation → Trip Details when Approved
def copy_allocation_to_trip(doc, method):

    if doc.status == "Approved":

        if not doc.driver_id or not doc.cab:
            frappe.throw(_("Please select Driver and Cab before approving."))

        doc.assigned_driver = doc.driver_id
        doc.assigned_cab = doc.cab


# 📧 Send Mail on Approved / Rejected
def send_status_mail(doc, method):

    employee_email = frappe.db.get_value(
        "Employee", doc.employee_id, "personal_email"
    )

    driver_email = None
    if doc.driver_id:
        driver_email = frappe.db.get_value("Driver", doc.driver_id, "email")

    # ✅ Approved
    if doc.status == "Approved":

        if employee_email:
            frappe.sendmail(
                recipients=[employee_email],
                subject="Cab Request Approved",
                message=f"""
                Hello {doc.employee_name},<br><br>
                Your cab request is approved.<br>
                Driver: {doc.assigned_driver}<br>
                Cab: {doc.assigned_cab}<br>
                Pickup: {doc.pickup_location}<br>
                Drop: {doc.drop_location}<br><br>
                Regards
                """,
            )

        if driver_email:
            frappe.sendmail(
                recipients=[driver_email],
                subject="New Cab Assignment",
                message=f"""
                You have been assigned a trip.<br><br>
                Employee: {doc.employee_name}<br>
                Pickup: {doc.pickup_location}<br>
                Drop: {doc.drop_location}<br>
                Contact: {doc.contact_no}
                """,
            )

    # ❌ Rejected
    if doc.status == "Rejected":

        if employee_email:
            frappe.sendmail(
                recipients=[employee_email],
                subject="Cab Request Rejected",
                message=f"""
                Hello {doc.employee_name},<br><br>
                Your cab request was rejected.<br>
                Reason: {doc.rejection_reason}<br><br>
                Regards
                """,
            )
            
@frappe.whitelist()
def get_employee_basic_details():
    employee = frappe.db.get_value(
        "Employee",
        {"user_id": frappe.session.user},
        ["name", "employee_name", "department", "mobile_no"],
        as_dict=True,
    )
    return employee

frappe.ui.form.on("Cab Request", {

    onload: function(frm) {

        // Auto set employee details for logged-in employee
        if (frappe.user.has_role("Employee") && frm.is_new()) {

            frappe.db.get_value("Employee", 
                { user_id: frappe.session.user },   // correct lookup
                ["name", "employee_name", "department"]
            ).then(r => {

                if (r.message) {
                    frm.set_value("employee_id", r.message.name);
                    frm.set_value("employee_name", r.message.employee_name);
                    frm.set_value("department", r.message.department);
                    frm.set_value("contact_no", r.message.mobile_no);
                    
                }

            });
        }
    },

    refresh: function(frm) {

        // Hide allocation fields for Employee
        if (frappe.user.has_role("Employee")) {
            frm.set_df_property("driver_id", "hidden", 0);
            frm.set_df_property("cab", "hidden", 1);
            frm.set_df_property("rejection_reason", "hidden", 1);
        }

        // Driver can only change status
        if (frappe.user.has_role("Driver")) {
            frm.set_df_property("driver_id", "read_only", 0);
            frm.set_df_property("cab", "read_only", 0);
        }
    }
});
